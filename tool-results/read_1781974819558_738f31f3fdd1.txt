   529→    @staticmethod
   530→    def _is_arabic_char(ch):
   531→        """فحص هل الحرف عربي"""
   532→        return bool(re.match(r'[\u0621-\u064A\u0660-\u0669\u06F0-\u06F9]', ch))
   533→
   534→    @staticmethod
   535→    def _is_latin_alpha(ch):
   536→        return ch.isascii() and ch.isalpha()
   537→
   538→    @staticmethod
   539→    def _is_url(text, pos):
   540→        """فحص هل الموضع داخل رابط"""
   541→        # تحقق بسيط - نبحث عن http:// أو https:// قبل الموضع
   542→        before = text[max(0, pos-30):pos]
   543→        return 'http://' in before or 'https://' in before or 't.me' in before or 'www.' in before
   544→
   545→    def _apply_combining(self, text, mark, skip_arabic_digits=False):
   546→        """إضافة علامة combining لكل حرف"""
   547→        result = []
   548→        for i, ch in enumerate(text):
   549→            # لا نضيف العلامة على المسافات وفواصل الأسطر
   550→            if ch in ' \n\t':
   551→                result.append(ch)
   552→                continue
   553→            # لا نضيف على الروابط (للحفاظ على روابط t.me)
   554→            if self._is_url(text, i):
   555→                result.append(ch)
   556→                continue
   557→            result.append(ch)
   558→            result.append(mark)
   559→        return ''.join(result)
   560→
   561→    def _apply_enclosing(self, text, mark):
   562→        """إحاطة كل حرف بعلامة enclosing"""
   563→        result = []
   564→        for i, ch in enumerate(text):
   565→            if ch in ' \n\t':
   566→                result.append(ch)
   567→                continue
   568→            if self._is_url(text, i):
   569→                result.append(ch)
   570→                continue
   571→            result.append(ch)
   572→            result.append(mark)
   573→        return ''.join(result)
   574→
   575→    def _apply_substitution(self, text, mapping):
   576→        """استبدال الحروف باستخدام جدول تعيين"""
   577→        result = []
   578→        for ch in text:
   579→            if ch in mapping:
   580→                result.append(mapping[ch])
   581→            else:
   582→                result.append(ch)
   583→        return ''.join(result)

   723→    def zalgo(self, text, intensity='medium'):
   724→        """
   725→        نص معطب (Glitch/Zalgo)
   726→        كدس علامات combining فوق وتحت كل حرف.
   727→        المصدر: https://fsymbols.com/generators/zalgo-text/
   728→        """
   729→        # تحديد شدة الزالجو
   730→        intensity_map = {
   731→            'light':  (1, 1, 0),  # (above, below, overlay)
   732→            'medium': (2, 2, 1),
   733→            'heavy':  (4, 4, 2),
   734→            'insane': (8, 8, 4),
   735→        }
   736→        n_above, n_below, n_overlay = intensity_map.get(intensity, intensity_map['medium'])
   737→
   738→        result = []
   739→        for i, ch in enumerate(text):
   740→            if ch in ' \n\t':
   741→                result.append(ch)
   742→                continue
   743→            if self._is_url(text, i):
   744→                result.append(ch)
   745→                continue
   746→            result.append(ch)
   747→            # كدس علامات فوق
   748→            for _ in range(n_above):
   749→                result.append(random.choice(ZALGO_ABOVE))
   750→            # كدس علامات تحت
   751→            for _ in range(n_below):
   752→                result.append(random.choice(ZALGO_BELOW))
   753→            # كدس علامات overlay
   754→            for _ in range(n_overlay):
   755→                result.append(random.choice(ZALGO_OVERLAY))
   756→        return ''.join(result)

   762→    def apply_style(self, text, style_name, **kwargs):
   763→        """
   764→        تطبيق نمط معين على النص
   765→        style_name: اسم النمط من STYLES
   766→        """
   767→        if not text or style_name not in self.STYLES:
   768→            return text
   769→
   770→        method = getattr(self, style_name, None)
   771→        if method is None:
   772→            return text
   773→
   774→        try:
   775→            # zalgo يحتاج معامل intensity
   776→            if style_name == 'zalgo':
   777→                intensity = kwargs.get('intensity', 'medium')
   778→                return method(text, intensity=intensity)
   779→            return method(text)
   780→        except Exception:
   781→            return text
