   302→   302→        q += 1

   304→   304→    if not nxt or not prev:
   305→   305→        return True                                   # لا يوجد زوج متصل
   306→   306→    if _is_arabic_letter(prev) and _is_arabic_letter(nxt):
   307→   307→        return prev in NON_FORWARD_JOINERS            # حرف غير واصل للأمام

   386→   386→                and not _is_combining_mark(chars[i + 1])

   395→   395→            ws, we = m.span()

   397→   397→            if any(ws <= f < we for f in forced):
   398→   398→                continue
   399→   399→            candidates = [
   400→   400→                i for i in range(ws, we - 1)
   401→   401→                if not _is_combining_mark(chars[i])
   402→   402→                and not _is_combining_mark(chars[i + 1])
   403→   403→            ]
   404→   404→            if candidates:
   405→   405→                forced.add(random.choice(candidates))
   406→   406→
   407→   407→    out = []
   408→   408→    consecutive = 0
   409→   409→
   410→   410→    for i, ch in enumerate(chars):
   411→   411→        out.append(ch)
   412→   412→
   413→   413→        if ch in (' ', '\n', '\t'):
   414→   414→            consecutive = 0
   415→   415→            continue
   416→   416→
   417→   417→        # لا ندرج حرفاً خفياً بين الحرف وتشكيله التالي (حركة/شدّة)

   419→   419→            continue

   423→   423→            out.append(random.choice(VS_POOL))
   424→   424→            consecutive += 1
   425→   425→            continue
   426→   426→

   428→   428→        local = density * (1.5 if i in boosted else 1.0)
   429→   429→        local = min(local, 0.80)
   430→   430→
   431→   431→        if consecutive < max_consecutive and random.random() < local:
   432→   432→            out.append(random.choice(VS_POOL))
   433→   433→            consecutive += 1
   434→   434→        else:
   435→   435→            consecutive = 0

   474→   474→def ensure_word_breaks(text: str, skip_spans=None) -> str:
   475→   475→    """
   476→   476→    🎯 ضمان نهائي للتجزئة (v4.1): يتأكد أن كل كلمة (3+ حروف) تحتوي
   477→   477→    حرفاً خفياً واحداً على الأقل حتى بعد الترقيق.
   478→   478→
   479→   479→    يُستدعى بعد _thin_invisibles لأن الترقيق قد يحذف الحرف المضمون
   480→   480→    في كلمات غير مفتاحية → هنا نعيده فوراً.
   481→   481→
   482→   482→    skip_spans: نطاقات لا تُمس (الرموز الحساسة قبل cloak - يجب أن تبقى
   483→   483→    نظيفة حتى يجدها cloak_sensitive_tokens لاحقاً)
   484→   484→    """
   485→   485→    if not text:
   486→   486→        return text
   487→   487→    skip = sorted(skip_spans or [])
   488→   488→
   489→   489→    def in_skip(pos: int) -> bool:
   490→   490→        for s, e in skip:
   491→   491→            if s <= pos < e:
   492→   492→                return True
   493→   493→        return False
   494→   494→
   495→   495→    chars = list(text)
   496→   496→    insertions = []  # (موضع الإدراج, الحرف)
   497→   497→    for m in re.finditer(r'\S{3,}', text):
   498→   498→        ws, we = m.span()
   499→   499→        if in_skip(ws) or in_skip(we - 1):
   500→   500→            continue
   501→   501→        seg = text[ws:we]
   502→   502→        has_ghost = any(ch in _GHOST_CHARS for ch in seg)
   503→   503→        if has_ghost:
   504→   504→            continue
   505→   505→        candidates = [
   506→   506→            i for i in range(ws, we - 1)
   507→   507→            if not _is_combining_mark(chars[i])
   508→   508→            and not _is_combining_mark(chars[i + 1])
   509→   509→        ]
   510→   510→        if candidates:
   511→   511→            pos = random.choice(candidates)
   512→   512→            insertions.append((pos + 1, random.choice(VS_POOL)))
   513→   513→    if not insertions:
   514→   514→        return text
   515→   515→    # الإدراج من النهاية للأمام حتى لا تتزحزح المواضع
   516→   516→    for pos, ch in sorted(insertions, reverse=True):
   517→   517→        chars.insert(pos, ch)
   518→   518→    return ''.join(chars)

   580→   580→def _thin_invisibles(text: str, max_ratio: float = 0.30) -> str:
   581→   581→    """
   582→   582→    ✂️ ترقيق تلقائي: إن تجاوزت نسبة الأحرف الشبحية الحد الآمن
   583→   583→    (يحفظنا من كشف بوتات الإحصاء الكمي للطابعات الخفية)
   584→   584→    نحذف الفائض عشوائياً مع:
   585→   585→    ✅ حماية الأحرف داخل الكلمات المفتاحية (لا تنفك التجزئة المضمونة)
   586→   586→    ✅ حماية بصمة النهاية (salt آخر 20 حرف)
   587→   587→    ✅ النص الظاهر لا يُمس أبداً
   588→   588→    """
   589→   589→    if not text:
   590→   590→        return text
   591→   591→    ratio = ghost_ratio(text)
   592→   592→    if ratio <= max_ratio:
   593→   593→        return text
   594→   594→
   595→   595→    chars = list(text)
   596→   596→
   597→   597→    # مواضع الكلمات المفتاحية محسوبة على النص الظاهر
   598→   598→    vis_positions = [i for i, ch in enumerate(chars) if ch not in _GHOST_CHARS]
   599→   599→    kw_visible_pos = set()
   600→   600→    if vis_positions:
   601→   601→        vis_str = ''.join(chars[i] for i in vis_positions)
   602→   602→        for s, e in _find_keyword_spans(vis_str):
   603→   603→            kw_visible_pos.update(range(s, e))
   604→   604→
   605→   605→    # ✂️ الحماية: الحرف الشبحي المجاور (يسار/يمين) لحرف كلمة مفتاحية
   606→   606→    # لا يُحذف حتى لا تنفك التجزئة المضمونة للكلمات المفتاحية
   607→   607→    removable = []
   608→   608→    for i, ch in enumerate(chars):
   609→   609→        if ch not in _GHOST_CHARS or i >= len(chars) - 20:
   610→   610→            continue  # البصمة الأخيرة محمية
   611→   611→        j = bisect.bisect_left(vis_positions, i) - 1  # آخر حرف ظاهر قبل i
   612→   612→        prev_in_kw = j >= 0 and j in kw_visible_pos
   613→   613→        next_in_kw = (j + 1 < len(vis_positions)) and ((j + 1) in kw_visible_pos)
   614→   614→        if prev_in_kw or next_in_kw:
   615→   615→            continue  # داخل/ملاصق لكلمة مفتاحية - محمي
   616→   616→        removable.append(i)
   617→   617→
   618→   618→    need = sum(1 for ch in chars if ch in _GHOST_CHARS) - int(max_ratio * len(chars))
   619→   619→    if need <= 0 or not removable:
   620→   620→        return text
   621→   621→
   622→   622→    random.shuffle(removable)
   623→   623→    remove_set = set(removable[:need])
   624→   624→    return ''.join(ch for i, ch in enumerate(chars) if i not in remove_set)
