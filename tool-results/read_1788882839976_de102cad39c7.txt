   241→   241→def ghost_vs_inject(text: str, density: float = 0.20,
   242→   242→                    keyword_boost: bool = True, max_consecutive: int = 2) -> str:
   243→   243→    """
   244→   244→    👻 حقن Variation Selectors داخل النص - القناة الحديثة v4.0
   245→   245→
   246→   246→    ✅ فئة Mn → Joining_Type = Transparent حسب معيار Unicode
   247→   247→       → اتصال الحروف العربية محفوظ 100% في كل محركات العرض
   248→   248→    ✅ يوضع في أي موضع حتى وسط الكلمة - لا يؤثر على الرسم أبداً
   249→   249→    ✅ 256 كودبونت → يكسر المطابقة التامة والـ n-gram والإحصاء الكمي
   250→   250→    ✅ لا يفصل الحرف عن تشكيله (حركة/شدّة تسبقه دائماً)
   251→   251→    🎯 ضمان التجزئة: كل كلمة مفتاحية تحتوي حرفاً خفياً واحداً على الأقل
   252→   252→       (إدخال قسري وليس احتمالياً) → مستحيل على بوتات المطابقة أن تجدها
   253→   253→    """
   254→   254→    if not text or density <= 0:
   255→   255→        return text
   256→   256→
   257→   257→    chars = list(text)
   258→   258→    boosted = set()
   259→   259→    forced = set()
   260→   260→    if keyword_boost:
   261→   261→        for s, e in _find_keyword_spans(text):
   262→   262→            boosted.update(range(s, e))
   263→   263→            # إدخال قسري مضمون بين حروف الكلمة نفسها (وليس بعدها!)
   264→   264→            # بين حرفين داخليين لا يلي أحدهما تشكيل حتى تتجزأ الكلمة فعلاً
   265→   265→            candidates = [
   266→   266→                i for i in range(s, max(s, e - 1))
   267→   267→                if not _is_combining_mark(chars[i])
   268→   268→                and not _is_combining_mark(chars[i + 1])
   269→   269→            ]
   270→   270→            if candidates:
   271→   271→                forced.add(random.choice(candidates))
   272→   272→
   273→   273→    out = []
   274→   274→    consecutive = 0
   275→   275→
   276→   276→    for i, ch in enumerate(chars):
   277→   277→        out.append(ch)
   278→   278→
   279→   279→        if ch in (' ', '\n', '\t'):
   280→   280→            consecutive = 0
   281→   281→            continue
   282→   282→
   283→   283→        # لا ندرج حرفاً خفياً بين الحرف وتشكيله التالي (حركة/شدّة)
   284→   284→        if i + 1 < len(chars) and _is_combining_mark(chars[i + 1]):
   285→   285→            continue
   286→   286→
   287→   287→        # 🎯 الإدخال القسري داخل الكلمات المفتاحية (ضمان التجزئة)
   288→   288→        if i in forced:
   289→   289→            out.append(random.choice(VS_POOL))
   290→   290→            consecutive += 1
   291→   291→            continue
   292→   292→
   293→   293→        # كثافة محلية مضاعفة داخل الكلمات المفتاحية
   294→   294→        local = density * (1.5 if i in boosted else 1.0)
   295→   295→        local = min(local, 0.80)
   296→   296→
   297→   297→        if consecutive < max_consecutive and random.random() < local:
   298→   298→            out.append(random.choice(VS_POOL))
   299→   299→            consecutive += 1
   300→   300→        else:
   301→   301→            consecutive = 0
   302→   302→
   303→   303→    return ''.join(out)
