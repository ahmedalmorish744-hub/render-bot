    86→def split_protected(text: str):

   190→def find_sensitive_spans(text: str):
   191→    """
   192→    مواضع الرموز الحساسة في النص [(بداية, نهاية)]
   193→    - يوزرات @username
   194→    - روابط http/https/t.me/wa.me/www
   195→    - أرقام هواتف (7+ رقم - غربي أو عربي)
   196→    """
   197→    spans = []
   198→    if not text:
   199→        return spans
   200→    for m in SENSITIVE_TOKEN_RE.finditer(text):
   201→        tok = m.group(0)
   202→        # فلترة الأرقام: هاتف حقيقي = 7 أرقام على الأقل
   203→        # (يستبعد التواريخ القصيرة والأسعار مثل 1500 أو 2026)
   204→        if _DIGITS_RE.match(tok) and '@' not in tok and '.' not in tok and '/' not in tok:
   205→            if len(_DIGITS_RE.findall(tok)) < 7:
   206→                continue
   207→        spans.append((m.start(), m.end()))
   208→    return spans

   211→def cloak_sensitive_tokens(text: str, density: float = 0.55) -> str:
   212→    """
   213→    🎯 الدرع الحساس: حقن VS17+ داخل الرموز الحساسة (يوزرات/أرقام/روابط)
   214→
   215→    قبل:  مرافق @ppppokl اتصل 0555123456
   216→    بعد:  نفس النص بالضبط للعين، لكن VS خفية متوزعة داخل كل رمز
   217→          → regex البوتات لا يجد @username ولا رقم هاتف ولا رابط
   218→
   219→    ✅ العين البشرية ترى نفس النص بالضبط (VS غير مرئية 100%)
   220→    ✅ بوتات الحماية تفشل في استخراج اليوزرات والأرقام والروابط
   221→    ⚖️ v4.1 كثافة 0.55 مع ضمانات: حرف VS واحد يكسر الـ regex كاملاً،
   222→       والتوزيع يبقي النسبة الإحصائية منخفضة ضد كشف الإحصاء الكمي
   223→    """
   224→    if not text:
   225→        return text
   226→    spans = find_sensitive_spans(text)
   227→    if not spans:
   228→        return text
   229→    result = []
   230→    last = 0
   231→    for s, e in spans:
   232→        result.append(text[last:s])
   233→        token = text[s:e]
   234→        tlen = len(token)
   235→        # مواضع القسمة الممكنة (بين الحروف)
   236→        gaps = list(range(len(token) - 1))
   237→        chosen = set()
   238→        # 🎯 ضمان 1: على الأقل حرفان خفيان في الرمز
   239→        if tlen >= 3 and gaps:
   240→            chosen.update(random.sample(gaps, min(2, len(gaps))))
   241→        # 🎯 ضمان 2: لا فجوة 3 أحرف متتالية بلا حرف خفي (توزيع متساوٍ)
   242→        for g in gaps:
   243→            if g % 3 == 1:
   244→                chosen.add(g)
   245→        # الباقي احتمالي
   246→        for g in gaps:
   247→            if g not in chosen and random.random() < density:
   248→                chosen.add(g)
   249→        for i, ch in enumerate(token):
   250→            result.append(ch)
   251→            if i in chosen:
   252→                result.append(random.choice(VS_POOL))
   253→        last = e
   254→    result.append(text[last:])
   255→    return ''.join(result)

   279→def _cf_safe_after(chars, i: int) -> bool:
   280→    """
   281→    هل وضعنا بعد الحرف i آمن لحرف Cf (غير واصل حسب المعيار)؟
   282→
   283→    القاعدة: يقطع الحرف غير الواصل الاتصال بين حرفين عربيين متجاورين
   284→    فقط إذا كان الحرف الأول يتصل للأمام والحرف الثاني يتصل من الخلف.
   285→    ⚠️ يجب تخطي الأحرف الشفافة (Mn/Me - ومنها VS المزرعة) على الجانبين
   286→    للوصول إلى الحرفين الحقيقيين المسؤولين عن الاتصال.
   287→    """
   288→    chars_len = len(chars)
   289→    if i + 1 >= chars_len:
   290→        return True                                   # نهاية النص
   291→    # لا نفصل الحرف عن تشكيله المباشر (حركة/شدّة/VS)
   292→    if _is_combining_mark(chars[i + 1]):
   293→        return False
   294→    # الحرف الحقيقي السابق (تخطي الشفاف)
   295→    p = i
   296→    while p >= 0 and _is_combining_mark(chars[p]):
   297→        p -= 1
   298→    prev = chars[p] if p >= 0 else ''
   299→    # الحرف الحقيقي التالي (تخطي الشفاف)
   300→    q = i + 1
   301→    while q < chars_len and _is_combining_mark(chars[q]):
   302→        q += 1
   303→    nxt = chars[q] if q < chars_len else ''
   304→    if not nxt or not prev:
   305→        return True                                   # لا يوجد زوج متصل
   306→    if _is_arabic_letter(prev) and _is_arabic_letter(nxt):
   307→        return prev in NON_FORWARD_JOINERS            # حرف غير واصل للأمام
   308→    return True                                       # لاتيني/أرقام/رموز/إيموجي

   367→    ✅ لا يفصل الحرف عن تشكيله (حركة/شدّة تسبقه دائماً)

   369→       واحداً مضموناً على الأقل (وليس كلمات الإعلانات فقط!) - لأن بوتات
   370→       الحماية تطابق أي قائمة كلمات، فالتجزئة يجب أن تشمل كل الكلمات

   373→        return text
   374→
   375→    chars = list(text)
   376→    boosted = set()
   377→    forced = set()
   378→    if keyword_boost:
   379→        for s, e in _find_keyword_spans(text):

   381→            # إدخال قسري مضمون بين حروف الكلمة نفسها (وليس بعدها!)
   382→            # بين حرفين داخليين لا يلي أحدهما تشكيل حتى تتجزأ الكلمة فعلاً
   383→            candidates = [
   384→                i for i in range(s, max(s, e - 1))
   385→                if not _is_combining_mark(chars[i])
   386→                and not _is_combining_mark(chars[i + 1])
   387→            ]
   388→            if candidates:
   389→                forced.add(random.choice(candidates))
   390→

   393→    if guarantee_word_break:

   395→            ws, we = m.span()

   397→            if any(ws <= f < we for f in forced):
   398→                continue
   399→            candidates = [
   400→                i for i in range(ws, we - 1)
   401→                if not _is_combining_mark(chars[i])
   402→                and not _is_combining_mark(chars[i + 1])
   403→            ]
   404→            if candidates:
   405→                forced.add(random.choice(candidates))
   406→
   407→    out = []
   408→    consecutive = 0
   409→
   410→    for i, ch in enumerate(chars):
   411→        out.append(ch)
   412→
   413→        if ch in (' ', '\n', '\t'):
   414→            consecutive = 0
   415→            continue
   416→
   417→        # لا ندرج حرفاً خفياً بين الحرف وتشكيله التالي (حركة/شدّة)
   418→        if i + 1 < len(chars) and _is_combining_mark(chars[i + 1]):
   419→            continue
   420→
   421→        # 🎯 الإدخال القسري داخل الكلمات المفتاحية (ضمان التجزئة)
   422→        if i in forced:
   423→            out.append(random.choice(VS_POOL))
   424→            consecutive += 1
   425→            continue

   432→            out.append(random.choice(VS_POOL))
   433→            consecutive += 1
   434→        else:
   435→            consecutive = 0
   436→
   437→    return ''.join(out)

   474→def ensure_word_breaks(text: str, skip_spans=None) -> str:
   475→    """
   476→    🎯 ضمان نهائي للتجزئة (v4.1): يتأكد أن كل كلمة (3+ حروف) تحتوي
   477→    حرفاً خفياً واحداً على الأقل حتى بعد الترقيق.
   478→
   479→    يُستدعى بعد _thin_invisibles لأن الترقيق قد يحذف الحرف المضمون
   480→    في كلمات غير مفتاحية → هنا نعيده فوراً.
   481→
   482→    skip_spans: نطاقات لا تُمس (الرموز الحساسة قبل cloak - يجب أن تبقى
   483→    نظيفة حتى يجدها cloak_sensitive_tokens لاحقاً)
   484→    """
   485→    if not text:
   486→        return text
   487→    skip = sorted(skip_spans or [])
   488→
   489→    def in_skip(pos: int) -> bool:
   490→        for s, e in skip:
   491→            if s <= pos < e:
   492→                return True
   493→        return False
   494→
   495→    chars = list(text)
   496→    insertions = []  # (موضع الإدراج, الحرف)
   497→    for m in re.finditer(r'\S{3,}', text):
   498→        ws, we = m.span()
   499→        if in_skip(ws) or in_skip(we - 1):
   500→            continue
   501→        seg = text[ws:we]
   502→        has_ghost = any(ch in _GHOST_CHARS for ch in seg)
   503→        if has_ghost:
   504→            continue
   505→        candidates = [
   506→            i for i in range(ws, we - 1)
   507→            if not _is_combining_mark(chars[i])
   508→            and not _is_combining_mark(chars[i + 1])
   509→        ]
   510→        if candidates:
   511→            pos = random.choice(candidates)
   512→            insertions.append((pos + 1, random.choice(VS_POOL)))
   513→    if not insertions:
   514→        return text
   515→    # الإدراج من النهاية للأمام حتى لا تتزحزح المواضع
   516→    for pos, ch in sorted(insertions, reverse=True):
   517→        chars.insert(pos, ch)
   518→    return ''.join(chars)

   580→def _thin_invisibles(text: str, max_ratio: float = 0.30) -> str:
   581→    """
   582→    ✂️ ترقيق تلقائي: إن تجاوزت نسبة الأحرف الشبحية الحد الآمن
   583→    (يحفظنا من كشف بوتات الإحصاء الكمي للطابعات الخفية)
   584→    نحذف الفائض عشوائياً مع:
   585→    ✅ حماية الأحرف داخل الكلمات المفتاحية (لا تنفك التجزئة المضمونة)
   586→    ✅ حماية بصمة النهاية (salt آخر 20 حرف)
   587→    ✅ النص الظاهر لا يُمس أبداً
   588→    """
   589→    if not text:
   590→        return text
   591→    ratio = ghost_ratio(text)
   592→    if ratio <= max_ratio:
   593→        return text
   594→
   595→    chars = list(text)
   596→
   597→    # مواضع الكلمات المفتاحية محسوبة على النص الظاهر
   598→    vis_positions = [i for i, ch in enumerate(chars) if ch not in _GHOST_CHARS]
   599→    kw_visible_pos = set()
   600→    if vis_positions:
   601→        vis_str = ''.join(chars[i] for i in vis_positions)
   602→        for s, e in _find_keyword_spans(vis_str):
   603→            kw_visible_pos.update(range(s, e))
   604→
   605→    # ✂️ الحماية: الحرف الشبحي المجاور (يسار/يمين) لحرف كلمة مفتاحية
   606→    # لا يُحذف حتى لا تنفك التجزئة المضمونة للكلمات المفتاحية
   607→    removable = []
   608→    for i, ch in enumerate(chars):
   609→        if ch not in _GHOST_CHARS or i >= len(chars) - 20:
   610→            continue  # البصمة الأخيرة محمية
   611→        j = bisect.bisect_left(vis_positions, i) - 1  # آخر حرف ظاهر قبل i
   612→        prev_in_kw = j >= 0 and j in kw_visible_pos
   613→        next_in_kw = (j + 1 < len(vis_positions)) and ((j + 1) in kw_visible_pos)
   614→        if prev_in_kw or next_in_kw:
   615→            continue  # داخل/ملاصق لكلمة مفتاحية - محمي
   616→        removable.append(i)
   617→
   618→    need = sum(1 for ch in chars if ch in _GHOST_CHARS) - int(max_ratio * len(chars))
   619→    if need <= 0 or not removable:
   620→        return text
   621→
   622→    random.shuffle(removable)
   623→    remove_set = set(removable[:need])
   624→    return ''.join(ch for i, ch in enumerate(chars) if i not in remove_set)

   737→            return self.enabled_layers[layer]
   738→        return None
   739→
   740→    def get_layer_status(self, layer: str) -> bool:
   741→        """حالة طبقة معينة (أسماء جديدة وقديمة مقبولة)"""
   742→        layer = self.LEGACY_ALIASES.get(layer, layer)
   743→        if layer in self.DISABLED_LAYERS:
   744→            return False
   745→        return self.enabled_layers.get(layer, False)

   747→    def _fit_length(self, text: str, budget: int = SAFE_LENGTH_BUDGET) -> str:
   748→        """
   749→        📏 حماية الطول: إن تجاوز النص ميزانية تيليجرام نحذف أحرفاً شبحية
   750→        (النص الظاهر لا يُمس أبداً) حتى نرجع تحت الحد.
   751→        ملاحظة: يُستدعى قبل الدرع الحساس (cloak) في الدورة الفعلية.
   752→        """
   753→        if len(text) <= budget:
   754→            return text
   755→        chars = [ch for ch in text if ch not in _GHOST_CHARS]
   756→        # لو النص الظاهر نفسه أطول من الحد فلا يمكننا المساس به
   757→        if len(chars) > budget:
   758→            return text
   759→        return ''.join(chars)

   761→    def obfuscate(self, text: str) -> Tuple[str, dict]:
   762→        """
   763→        تطبيق Ghost Encoding على النص
   764→
   765→        Returns:
   766→            (النص المشفر, معلومات الطبقات المطبقة)
   767→        """
   768→        if not text:
   769→            return text, {'layers': []}
   770→
   771→        self.stats['total_messages'] += 1
   772→        applied_layers = []
   773→
   774→        # 🧹 الخطوة 0: تنظيف الأحرف الخطرة + إصلاح تلوث أشكال العرض
   775→        result = sanitize_invisible_chars(text)
   776→
   777→        # 🛡️ حماية التكرار: نص مشفر مسبقاً؟ نظف الأحرف الشبحية أولاً
   778→        # (سبب "تلبّط بدء النشر" سابقاً كان تكديس التحويلات - الآن مستحيل)
   779→        if ghost_ratio(result) > 0.12:
   780→            result = strip_ghost_chars(result)
   781→            self.stats['double_encode_saved'] += 1
   782→
   783→        # 🎯 الخطوة 0.5 (v4.1): تقسيم النص إلى مقاطع عادية ورموز حساسة
   784→        # (اليوزرات/الهواتف/الروابط) - الطبقات تُطبق على المقاطع العادية فقط
   785→        # ثم تُعاد الرموز الحساسة بتكويد VS بين كل حرفين (درع اليوزرات)
   786→        spans = find_sensitive_spans(result)
   787→        segments = []  # [(نص, حساس؟)]
   788→        last = 0
   789→        for s, e in spans:
   790→            if s > last:
   791→                segments.append((result[last:s], False))
   792→            segments.append((result[s:e], True))
   793→            last = e
   794→        if last < len(result):
   795→            segments.append((result[last:], False))
   796→
   797→        vs_on = self.enabled_layers['ghost_vs_channel'] and self.config['vs_density'] > 0
   798→        cf_on = self.enabled_layers['cf_boundary'] and self.config['cf_density'] > 0
   799→
   800→        processed = []
   801→        for seg, is_sensitive in segments:
   802→            if is_sensitive:
   803→                processed.append(seg)  # الرمز الحساس يمر نظيفاً الآن - يُكوَّد في النهاية
   804→                continue
   805→            out = seg
   806→            # 1️⃣ Ghost VS Channel - قناة VS الشفافة للتشكيل (الأساسية)
   807→            if vs_on:
   808→                out = ghost_vs_inject(out, self.config['vs_density'], True)
   809→            # 2️⃣ CF Boundary - أحرف حدودية آمنة (تنويع القنوات)
   810→            if cf_on:
   811→                out = cf_boundary_inject(out, self.config['cf_density'])
   812→            processed.append(out)
   813→        if vs_on:
   814→            applied_layers.append('ghost_vs_channel')
   815→        if cf_on:
   816→            applied_layers.append('cf_boundary')
   817→
   818→        result = ''.join(processed)
   819→
   820→        # ✂️ ترقيق تلقائي - يبقي نسبة الأحرف الخفية تحت رادار الكشف الإحصائي
   821→        # (مع حماية تجزئة الكلمات المفتاحية وبصمة النهاية)
   822→        result = _thin_invisibles(result, max_ratio=0.25)
   823→
   824→        # 🎯 ضمان نهائي للتجزئة: الترقيق قد يحذف حرفاً مضموناً → نعيده
   825→        # (نكتشف الرموز الحساسة على النص الحالي حتى تكون المواضع دقيقة،
   826→        #  ونمسها حتى تبقى نظيفة للدرع لاحقاً)
   827→        result = ensure_word_breaks(result, skip_spans=find_sensitive_spans(result))
   828→
   829→        # 3️⃣ Keyword Boost - مطبق داخلياً في قناة VS (توثيق فقط هنا)
   830→        if self.enabled_layers['keyword_boost']:
   831→            applied_layers.append('keyword_boost')
   832→
   833→        # 4️⃣ Salt - بصمة فريدة لكل رسالة
   834→        if self.enabled_layers['salt']:
   835→            result = add_anti_similarity_salt(result)
   836→            applied_layers.append('salt')
   837→
   838→        # 📏 حماية الطول (قبل الدرع الحساس - الدرع يضيف حروفاً قليلة فقط)
   839→        result = self._fit_length(result)
   840→
   841→        # 🎯 الخطوة 5 (v4.1): درع اليوزرات والأرقام والروابط (بعد كل شيء
   842→        # حتى لا يلمسه الترقيق أو حماية الطول)
   843→        if spans:
   844→            result = cloak_sensitive_tokens(result)
   845→            applied_layers.append('sensitive_shield')
   846→
   847→        self.stats['total_obfuscated'] += 1
   848→
   849→        invisible_count = sum(1 for ch in result if ch in _GHOST_CHARS)
   850→
   851→        return result, {
   852→            'layers': applied_layers,
   853→            'profile': self.profile,
   854→            'length_before': len(text),
   855→            'length_after': len(result),
   856→            'invisible_chars': invisible_count,
   857→        }
