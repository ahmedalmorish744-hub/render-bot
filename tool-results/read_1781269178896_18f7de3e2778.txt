   312→class UltimateAntiDetection:
   313→    """
   314→    نظام مكافحة الكشف المتطور - النص يبقى كما هو للمستخدم العادي
   315→    الروابط والمعرفات تبقى قابلة للنقر تماماً
   316→    التغييرات غير مرئية للعين المجردة فقط:
   317→    - أحرف غير مرئية بين الكلمات (ليس داخل الروابط)
   318→    - مسافات بديلة بدل المسافة العادية (ليس داخل الروابط)
   319→    - حروف لاتينية متشابهة (homoglyphs) بنسبة منخفضة (ليس داخل الروابط)
   320→    - الروابط والمعرفات تترك كما هي تماماً بدون أي تعديل
   321→    """
   322→    def __init__(self):
   323→        self.invisible_chars = ['\u200B', '\u200C', '\u200D', '\uFEFF']
   324→        self.homoglyphs = {
   325→            'a': '\u0430', 'e': '\u0435', 'o': '\u043E',
   326→            'c': '\u0441', 'p': '\u0440', 'x': '\u0445',
   327→            'i': '\u0456', 'j': '\u0458',
   328→        }
   329→        self.sent_messages_cache = deque(maxlen=500)
   330→
   331→    def _extract_protected_segments(self, text):
   332→        """استخراج الروابط والمعرفات لحمايتها من أي تعديل"""
   333→        protected = []
   334→        for match in re.finditer(r'https?://\S+', text):
   335→            protected.append((match.start(), match.end(), match.group()))
   336→        for match in re.finditer(r'@[a-zA-Z0-9_]{3,}', text):
   337→            overlaps = any(match.start() >= p[0] and match.start() < p[1] for p in protected)
   338→            if not overlaps:
   339→                protected.append((match.start(), match.end(), match.group()))
   340→        protected.sort(key=lambda x: x[0])
   341→        clean = []
   342→        for seg in protected:
   343→            if not clean:
   344→                clean.append(seg)
   345→            elif seg[0] >= clean[-1][1]:
   346→                clean.append(seg)
   347→        return clean
   348→
   349→    def _split_text_protected(self, text):
   350→        """تقسيم النص إلى أجزاء محمية (روابط/معرفات) وأجزاء عادية"""
   351→        protected = self._extract_protected_segments(text)
   352→        segments = []
   353→        last_end = 0
   354→        for start, end, original in protected:
   355→            if start > last_end:
   356→                segments.append(('text', text[last_end:start]))
   357→            segments.append(('protected', original))
   358→            last_end = end
   359→        if last_end < len(text):
   360→            segments.append(('text', text[last_end:]))
   361→        return segments
   362→
   363→    def _apply_to_text_only(self, text, func):
   364→        """تطبيق دالة تحويل على النص العادي فقط بدون الروابط والمعرفات"""
   365→        segments = self._split_text_protected(text)
   366→        result = []
   367→        for seg_type, seg_text in segments:
   368→            if seg_type == 'protected':
   369→                result.append(seg_text)  # لا نعدل الروابط والمعرفات أبداً
   370→            else:
   371→                result.append(func(seg_text))
   372→        return ''.join(result)
   373→
   374→    def apply_homoglyphs(self, text, intensity=0.06):
   375→        def _apply(seg):
   376→            result = []
   377→            for char in seg:
   378→                if char in self.homoglyphs and random.random() < intensity:
   379→                    result.append(self.homoglyphs[char])
   380→                else:
   381→                    result.append(char)
   382→            return ''.join(result)
   383→        return self._apply_to_text_only(text, _apply)
   384→
   385→    def add_invisible_between_words(self, text, intensity=0.15):
   386→        def _apply(seg):
   387→            if not seg or len(seg) < 5:
   388→                return seg
   389→            result = list(seg)
   390→            space_positions = [i for i, c in enumerate(result) if c == ' ']
   391→            if not space_positions:
   392→                return seg
   393→            insert_positions = []
   394→            for pos in space_positions:
   395→                if random.random() < intensity:
   396→                    insert_positions.append(pos + 1)
   397→            for pos in reversed(insert_positions):
   398→                inv = random.choice(self.invisible_chars)
   399→                result.insert(pos, inv)
   400→            return ''.join(result)
   401→        return self._apply_to_text_only(text, _apply)
   402→
   403→    def replace_some_spaces(self, text, intensity=0.3):
   404→        def _apply(seg):
   405→            result = list(seg)
   406→            space_positions = [i for i, c in enumerate(result) if c == ' ']
   407→            for pos in space_positions:
   408→                if random.random() < intensity:
   409→                    replacement = random.choice(['\u00A0', '\u2009', '\u202F'])
   410→                    result[pos] = replacement
   411→            return ''.join(result)
   412→        return self._apply_to_text_only(text, _apply)
   413→
   414→    def is_duplicate_message(self, text, group_id):
   415→        cache_key = f"{group_id}:{hash(text)}"
   416→        if cache_key in self.sent_messages_cache:
   417→            return True
   418→        self.sent_messages_cache.append(cache_key)
   419→        return False
   420→
   421→    def generate_ultimate_variation(self, text, group_id=None):
   422→        if get_setting('anti_detect', 'on') != 'on':
   423→            return text
   424→        if group_id and self.is_duplicate_message(text, group_id):
   425→            text = '\u200B' + text
   426→        result = text
   427→        # الروابط والمعرفات لا يتم تعديلها أبداً - تبقى قابلة للنقر
   428→        result = self.apply_homoglyphs(result, intensity=0.06)
   429→        result = self.replace_some_spaces(result, intensity=0.3)
   430→        result = self.add_invisible_between_words(result, intensity=0.15)
   431→        return result

   980→    def _apply_arabic_smart_obfuscation(self, text, intensity=0.35):
   981→        """تطبيق تشويش عربي ذكي يحافظ على المقروئية تماماً
   982→        
   983→        تقنيات:
   984→        1. تحويل النماذج المركبة إلى مفككة (NFC → NFD) - يبدو متطابقاً مرئياً
   985→        2. أرقام عربية-هندية بدل اللاتينية
   986→        3. أحرف غير مرئية استراتيجية (متعددة الأنواع)
   987→        4. مسافات بديلة متنوعة
   988→        5. علامات RTL/ALM خفية
   989→        """
   990→        if not text:
   991→            return text
   992→        
   993→        # ═══ 1. NFD Decomposition - يبدو متطابقاً لكن الكود مختلف ═══
   994→        result = []
   995→        for c in text:
   996→            if c in self.ARABIC_DECOMPOSE_MAP and random.random() < intensity:
   997→                base, combining = self.ARABIC_DECOMPOSE_MAP[c]
   998→                result.append(base)
   999→                result.append(combining)
  1000→            else:
  1001→                result.append(c)
  1002→        text = ''.join(result)
  1003→        
  1004→        # ═══ 2. Arabic-Indic digits (أرقام مفردة فقط وليس أرقام هواتف) ═══
  1005→        result = list(text)
  1006→        prev_was_digit = False
  1007→        for i, c in enumerate(result):
  1008→            if c.isdigit() and c in self.ARABIC_DIGIT_MAP:
  1009→                if prev_was_digit:
  1010→                    prev_was_digit = True
  1011→                    continue
  1012→                if random.random() < 0.25:
  1013→                    digit_map = random.choice([self.ARABIC_DIGIT_MAP, self.ARABIC_EXTENDED_DIGIT_MAP])
  1014→                    result[i] = digit_map[c]
  1015→                prev_was_digit = True
  1016→            else:
  1017→                prev_was_digit = False
  1018→        text = ''.join(result)
  1019→        
  1020→        # ═══ 3. أحرف غير مرئية احترافية (متعددة الأنواع) ═══
  1021→        inv_chars = ['\u200B', '\u200C', '\u200D', '\uFEFF', '\u2060', '\u061C']
  1022→        
  1023→        # حرف غير مرئي في البداية
  1024→        text = random.choice(inv_chars[:4]) + text
  1025→        
  1026→        # أحرف غير مرئية بين الكلمات العربية
  1027→        words = text.split(' ')
  1028→        new_words = []
  1029→        for i, w in enumerate(words):
  1030→            new_words.append(w)
  1031→            if i < len(words) - 1 and random.random() < 0.2:
  1032→                new_words.append(random.choice(inv_chars[:4]))
  1033→        text = ' '.join(new_words)
  1034→        
  1035→        # أحرف غير مرئية بعد علامات الترقيم العربية
  1036→        arabic_punct = '،.؛:!؟-'
  1037→        result = list(text)
  1038→        insert_positions = []
  1039→        for i, c in enumerate(result):
  1040→            if c in arabic_punct and random.random() < 0.25:
  1041→                insert_positions.append((i + 1, random.choice(inv_chars[:4])))
  1042→        for pos, char in sorted(insert_positions, key=lambda x: x[0], reverse=True):
  1043→            result.insert(pos, char)
  1044→        text = ''.join(result)
  1045→        
  1046→        # ═══ 4. مسافات بديلة متنوعة ═══
  1047→        alt_spaces = ['\u00A0', '\u2009', '\u202F', '\u2007', '\u2006', '\u2005']
  1048→        result = list(text)
  1049→        for i, c in enumerate(result):
  1050→            if c == ' ' and random.random() < 0.3:
  1051→                result[i] = random.choice(alt_spaces)
  1052→        text = ''.join(result)
  1053→        
  1054→        # ═══ 5. علامات RTL/ALM خفية ═══
  1055→        if random.random() < 0.2:
  1056→            text = text + '\u200F'  # RTL Mark
  1057→        if random.random() < 0.1:
  1058→            text = '\u061C' + text  # Arabic Letter Mark at start
  1059→        
  1060→        # حرف غير مرئي في النهاية
  1061→        text = text + random.choice(inv_chars[:4])
  1062→        
  1063→        return text

  1663→        i = 0

  1669→                    new_result.append(base)
  1670→                    new_result.append(combining)
  1671→                else:
  1672→                    new_result.append(c)
  1673→            else:
  1674→                new_result.append(c)
