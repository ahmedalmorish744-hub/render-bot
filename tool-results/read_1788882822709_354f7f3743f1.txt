    81→def split_protected(text: str):

    83→    parts = []

    86→        if m.start() > last:
    87→            parts.append((text[last:m.start()], False))
    88→        parts.append((m.group(0), True))
    89→        last = m.end()

    91→        parts.append((text[last:], False))
    92→    return parts if parts else [(text, False)]

   168→def _cf_safe_after(chars, i: int) -> bool:
   169→    """
   170→    هل وضعنا بعد الحرف i آمن لحرف Cf (غير واصل حسب المعيار)؟
   171→
   172→    القاعدة: يقطع الحرف غير الواصل الاتصال بين حرفين عربيين متجاورين
   173→    فقط إذا كان الحرف الأول يتصل للأمام والحرف الثاني يتصل من الخلف.
   174→    ⚠️ يجب تخطي الأحرف الشفافة (Mn/Me - ومنها VS المزرعة) على الجانبين
   175→    للوصول إلى الحرفين الحقيقيين المسؤولين عن الاتصال.
   176→    """
   177→    chars_len = len(chars)
   178→    if i + 1 >= chars_len:
   179→        return True                                   # نهاية النص
   180→    # لا نفصل الحرف عن تشكيله المباشر (حركة/شدّة/VS)
   181→    if _is_combining_mark(chars[i + 1]):
   182→        return False
   183→    # الحرف الحقيقي السابق (تخطي الشفاف)
   184→    p = i
   185→    while p >= 0 and _is_combining_mark(chars[p]):
   186→        p -= 1
   187→    prev = chars[p] if p >= 0 else ''
   188→    # الحرف الحقيقي التالي (تخطي الشفاف)
   189→    q = i + 1
   190→    while q < chars_len and _is_combining_mark(chars[q]):
   191→        q += 1
   192→    nxt = chars[q] if q < chars_len else ''
   193→    if not nxt or not prev:
   194→        return True                                   # لا يوجد زوج متصل
   195→    if _is_arabic_letter(prev) and _is_arabic_letter(nxt):
   196→        return prev in NON_FORWARD_JOINERS            # حرف غير واصل للأمام
   197→    return True                                       # لاتيني/أرقام/رموز/إيموجي

   211→            if idx == -1:
   212→                break
   213→            spans.append((idx, idx + len(kw)))
   214→            start = idx + 1
   215→    return spans

   241→def ghost_vs_inject(text: str, density: float = 0.20,
   242→                    keyword_boost: bool = True, max_consecutive: int = 2) -> str:
   243→    """
   244→    👻 حقن Variation Selectors داخل النص - القناة الحديثة v4.0
   245→
   246→    ✅ فئة Mn → Joining_Type = Transparent حسب معيار Unicode
   247→       → اتصال الحروف العربية محفوظ 100% في كل محركات العرض
   248→    ✅ يوضع في أي موضع حتى وسط الكلمة - لا يؤثر على الرسم أبداً
   249→    ✅ 256 كودبونت → يكسر المطابقة التامة والـ n-gram والإحصاء الكمي
   250→    ✅ لا يفصل الحرف عن تشكيله (حركة/شدّة تسبقه دائماً)
   251→    🎯 ضمان التجزئة: كل كلمة مفتاحية تحتوي حرفاً خفياً واحداً على الأقل
   252→       (إدخال قسري وليس احتمالياً) → مستحيل على بوتات المطابقة أن تجدها
   253→    """
   254→    if not text or density <= 0:
   255→        return text
   256→
   257→    chars = list(text)
   258→    boosted = set()
   259→    forced = set()
   260→    if keyword_boost:
   261→        for s, e in _find_keyword_spans(text):
   262→            boosted.update(range(s, e))
   263→            # إدخال قسري مضمون بين حروف الكلمة نفسها (وليس بعدها!)
   264→            # بين حرفين داخليين لا يلي أحدهما تشكيل حتى تتجزأ الكلمة فعلاً
   265→            candidates = [
   266→                i for i in range(s, max(s, e - 1))
   267→                if not _is_combining_mark(chars[i])
   268→                and not _is_combining_mark(chars[i + 1])
   269→            ]
   270→            if candidates:
   271→                forced.add(random.choice(candidates))
   272→
   273→    out = []
   274→    consecutive = 0
   275→
   276→    for i, ch in enumerate(chars):
   277→        out.append(ch)
   278→
   279→        if ch in (' ', '\n', '\t'):
   280→            consecutive = 0
   281→            continue
   282→
   283→        # لا ندرج حرفاً خفياً بين الحرف وتشكيله التالي (حركة/شدّة)
   284→        if i + 1 < len(chars) and _is_combining_mark(chars[i + 1]):
   285→            continue
   286→
   287→        # 🎯 الإدخال القسري داخل الكلمات المفتاحية (ضمان التجزئة)
   288→        if i in forced:
   289→            out.append(random.choice(VS_POOL))
   290→            consecutive += 1
   291→            continue
   292→
   293→        # كثافة محلية مضاعفة داخل الكلمات المفتاحية
   294→        local = density * (1.5 if i in boosted else 1.0)
   295→        local = min(local, 0.80)
   296→
   297→        if consecutive < max_consecutive and random.random() < local:
   298→            out.append(random.choice(VS_POOL))
   299→            consecutive += 1
   300→        else:
   301→            consecutive = 0
   302→
   303→    return ''.join(out)

   310→def cf_boundary_inject(text: str, density: float = 0.10) -> str:
   311→    """
   312→    🧱 حقن U+2063/U+2064 (Cf) في المواضع الحيادية للاتصال فقط
   313→
   314→    ✅ نهايات الكلمات + بعد الحروف غير الواصلة للأمام (اأإآدذرزوؤةء)
   315→    ✅ لا يوضع أبداً وسط الكلمة بعد حرف واصل → لا يقطع الاتصال
   316→       حتى في المحركات الصارمة التي تتبع المعيار حرفياً
   317→    ✅ تنويع القنوات: لو نظّفت بوتات الحماية فئة معينة تبقى القناة الأخرى
   318→    """
   319→    if not text or density <= 0:
   320→        return text
   321→
   322→    chars = list(text)
   323→    out = []
   324→
   325→    for i, ch in enumerate(chars):
   326→        out.append(ch)
   327→
   328→        if ch in (' ', '\n', '\t'):
   329→            continue
   330→
   331→        if i + 1 < len(chars) and _is_combining_mark(chars[i + 1]):
   332→            continue
   333→
   334→        if random.random() < density and _cf_safe_after(chars, i):
   335→            out.append(random.choice(INVISIBLE_SHIELD_CHARS))

   363→
   364→    ✅ يكسر كشف الرسائل المكررة (dedup) والهاش المتطابق
   365→    ✅ كل رسالة في كل مجموعة لها بصمة مختلفة عشوائياً
   366→    ✅ يستخدم قناة VS فقط → لا يتعارض مع أبجدية بصمة stego
   367→    ✅ في نهاية الرسالة → صفر تأثير على النص أو رسمه
   368→    ✅ الطول يتناسب مع طول الرسالة (حتى لا ترتفع نسبة الخفي
   369→       في الرسائل القصيرة فوق رادار الكشف الإحصائي)
   370→    """
   371→    if not text:
   372→        return text
   373→    salt_len = min(random.randint(8, 14), max(4, len(text) // 10))
   374→    salt = ''.join(random.choice(VS_POOL) for _ in range(salt_len))
   375→    return text + salt
   376→
   377→# ═══════════════════════════════════════════════════════════════

   379→# ═══════════════════════════════════════════════════════════════
   380→
   381→_GHOST_CHARS = set(VS_POOL) | set(INVISIBLE_SHIELD_CHARS)

   384→def strip_ghost_chars(text: str) -> str:
   385→    """إزالة كل الأحرف الشبحية (VS + Cf الحدودية) وإرجاع النص الظاهر الخالص"""
   386→    if not text:
   387→        return text
   388→    return ''.join(ch for ch in text if ch not in _GHOST_CHARS)

   391→def ghost_ratio(text: str) -> float:
   392→    """نسبة الأحرف الشبحية في النص (للكشف عن نص مشفر مسبقاً)"""
   393→    if not text:
   394→        return 0.0
   395→    count = sum(1 for ch in text if ch in _GHOST_CHARS)
   396→    return count / max(1, len(text))

   398→
   399→def _thin_invisibles(text: str, max_ratio: float = 0.30) -> str:
   400→    """
   401→    ✂️ ترقيق تلقائي: إن تجاوزت نسبة الأحرف الشبحية الحد الآمن
   402→    (يحفظنا من كشف بوتات الإحصاء الكمي للطابعات الخفية)
   403→    نحذف الفائض عشوائياً مع:
   404→    ✅ حماية الأحرف داخل الكلمات المفتاحية (لا تنفك التجزئة المضمونة)
   405→    ✅ حماية بصمة النهاية (salt آخر 20 حرف)
   406→    ✅ النص الظاهر لا يُمس أبداً
   407→    """
   408→    if not text:
   409→        return text
   410→    ratio = ghost_ratio(text)
   411→    if ratio <= max_ratio:
   412→        return text
   413→
   414→    chars = list(text)
   415→
   416→    # مواضع الكلمات المفتاحية محسوبة على النص الظاهر
   417→    vis_positions = [i for i, ch in enumerate(chars) if ch not in _GHOST_CHARS]
   418→    kw_visible_pos = set()
   419→    if vis_positions:
   420→        vis_str = ''.join(chars[i] for i in vis_positions)
   421→        for s, e in _find_keyword_spans(vis_str):
   422→            kw_visible_pos.update(range(s, e))
   423→
   424→    # ✂️ الحماية: الحرف الشبحي المجاور (يسار/يمين) لحرف كلمة مفتاحية
   425→    # لا يُحذف حتى لا تنفك التجزئة المضمونة للكلمات المفتاحية
   426→    removable = []
   427→    for i, ch in enumerate(chars):
   428→        if ch not in _GHOST_CHARS or i >= len(chars) - 20:
   429→            continue  # البصمة الأخيرة محمية
   430→        j = bisect.bisect_left(vis_positions, i) - 1  # آخر حرف ظاهر قبل i
   431→        prev_in_kw = j >= 0 and j in kw_visible_pos
   432→        next_in_kw = (j + 1 < len(vis_positions)) and ((j + 1) in kw_visible_pos)
   433→        if prev_in_kw or next_in_kw:
   434→            continue  # داخل/ملاصق لكلمة مفتاحية - محمي
   435→        removable.append(i)
   436→
   437→    need = sum(1 for ch in chars if ch in _GHOST_CHARS) - int(max_ratio * len(chars))
   438→    if need <= 0 or not removable:
   439→        return text
   440→
   441→    random.shuffle(removable)
   442→    remove_set = set(removable[:need])
   443→    return ''.join(ch for i, ch in enumerate(chars) if i not in remove_set)
   444→

   449→
   450→def apply_arabic_presentation_forms(text: str, intensity: float = 0.0) -> str:
